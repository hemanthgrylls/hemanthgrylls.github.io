  \draw[black, thick] (0,0) circle(1.25ex);
  \fill[#1] (0,0) -- (1.25ex,0) arc (0:180:1.25ex) -- cycle;

\newcommand{\tricolorcircle}[3]{

  \fill[#1] (0,0) -- (1.25ex,0) arc (0:120:1.25ex) -- cycle;
  \fill[#2] (0,0) -- ({1.25ex*cos(120)},{1.25ex*sin(120)}) arc (120:240:1.25ex) -- cycle;
  \fill[#3] (0,0) -- ({1.25ex*cos(240)},{1.25ex*sin(240)}) arc (240:360:1.25ex) -- cycle;
}}
