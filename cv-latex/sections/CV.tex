\noindent \textbf{Johns Hopkins University, Department of Mechanical Engineering} \hfill Baltimore, MD, USA\\
\noindent \textit{As postdoctoral fellow} \hfill Advisor: Prof. Tamer A. Zaki \\[-5mm]

\begin{itemize}[itemsep=0pt, topsep=5pt, label=\halfcircle{lightred}{green}]
    \item \textbf{Data Assimilation for High-Speed Flows} (Ongoing): Development of a feature-preserving latent-space ensemble Kalman filter (Latent-EnKF) for assimilating sparse observations into shock-containing flows, preserving discontinuities that conventional latent representations smear.
\end{itemize}
\begin{itemize}[itemsep=0pt, topsep=0pt, label=\tricolorcircle{lightred}{cyan}{yellow}]
    \item \textbf{Differentiable Solvers and Adjoint Sensitivity} (Ongoing): Adjoint-based sensitivity analysis for compressible flow solvers, and direct simulation Monte Carlo (DSMC) for rarefied and transitional-regime flows. \\
\end{itemize}

\noindent \textbf{Technion, Faculty of Mechanical Engineering} \hfill Haifa, Israel\\
\noindent \textit{As postdoctoral fellow} \hfill Advisor: Prof. Steven H. Frankel \\[-5mm]

\begin{itemize}[itemsep=0pt, topsep=5pt, label=\halfcircle{lightred}{yellow}]
    \item \textbf{Scramjet Un-start Phenomenon:} Development of in-house tools for dynamic mesh simulation to study the un-start phenomenon prevalent in scramjet engines using GPU-accelerated large eddy simulations.
\end{itemize}
\begin{itemize}[itemsep=0pt, topsep=0pt, label=\colorcircle{green}]
    \item \textbf{Implicit Neural Representation Networks:} Proposed SOS-SIREN, a \textit{noisy sinusoidal representation neural network} for reconstructing complex audio and image signals with dominant high-frequency features as continuous neural functions. Applications include compressing large fluid flow datasets and solving forward problems via Physics Informed Neural Networks.
\end{itemize}
\begin{itemize}[itemsep=0pt, topsep=0pt, label=\tricolorcircle{lightred}{cyan}{green}]
    \item \textbf{Method and Solver Development:} \textbf{(a)} Development of a neural network based optimal order detection model for high-resolution compressible flow simulations using polynomials of different orders and spectral resolutions. \textbf{(b)} Development of a new unstructured flux reconstruction (nodal Discontinuous Galerkin) based compressible Navier-Stokes solver with artificial diffusion based shock-capturing capability.
\end{itemize}
\begin{itemize}[itemsep=0pt, topsep=0pt, label=\colorcircle{lightred}]
    \item \textbf{Shock Wave Dynamics:} Collaborative work with Prof. Omri Ram and the TFML laboratory at Technion on investigating the dynamics of shock waves passing through roughness elements in confined tubes. \\
\end{itemize}

\noindent \textit{As graduate student} (Ph.D.) \hfill Advisor: Prof. Steven H. Frankel \\[-5mm]
\begin{itemize}[itemsep=0pt, topsep=5pt, label=\halfcircle{cyan}{cyan}]
    \item \textbf{Method Development:} \textbf{(a)} Proposed a new resolution-optimized high-order \textit{dispersion-relation-preserving} scheme for non-linear diffusion terms; demonstrated state-of-the-art accuracy and improved stability. \textbf{(b)} Investigated the use of spatially \textit{adaptive modal filtering} for resolving discontinuities in compressible flows through the flux reconstruction framework.
\end{itemize}
\begin{itemize}[itemsep=0pt, topsep=0pt, label=\halfcircle{yellow}{yellow}]
    \item \textbf{Solver Development (from scratch):}  
    \textbf{(a) MIGHTY:} Built a multi-GPU accelerated, curvilinear multi-block Navier-Stokes solver with state-of-the-art shock-capturing and diffusion schemes.  
    \textbf{(b) CAF-FR:} Developed from scratch a Flux reconstruction solver for Euler equations with shock capturing capability.  
    \textbf{(c) MIGHTY-KEEP:} Co-developed a kinetic energy and entropy-preserving compressible Navier-Stokes solver.  
    \textbf{(d) LEGOs:} Wrote a Fortran-based grid generator for simple multi-block geometries.  
\end{itemize}
\begin{itemize}[itemsep=0pt, topsep=0pt, label=\halfcircle{green}{green}]
    \item \textbf{Physics Informed Machine Learning:} One of the lead developers of PINNACLE - a GPU-accelerated physics informed neural network (PINNs) framework to solve forward and inverse problems relevant to Engineering.
\end{itemize}
\begin{itemize}[itemsep=0pt, topsep=0pt, label=\halfcircle{lightred}{lightred}]
    \item \textbf{Supersonic Jet Noise}: Conducted Large Eddy Simulations to study the influence of jet temperature on shock-cell spacing and shear-layer convection velocity at a Reynolds number \( O(10^5) \). Performed modal analysis of screech dynamics across varying temperature ratios. Investigated the influence of impingement wall curvature on flow unsteadiness and screech characteristics for convex and concave impinging wall geometries.  \\ 
\end{itemize}

\noindent \textbf{Indian Institute of Technology Madras} \hfill Chennai, India\\
\noindent \textit{As graduate student} (M.Tech.) \hfill Advisors: R. Sriram \\[-5mm]
\begin{itemize}[itemsep=0pt, topsep=5pt, label=\halfcircle{lightred}{lightred}]
    \item \textbf{3D Shock-Boundary Layer Interaction and Control:} Conducted wind tunnel experiments to analyze separation length scaling laws in protuberance-induced shock-boundary layer interactions across various geometries. Successfully demonstrated separation and unsteadiness reduction through \textit{boundary layer bleed} via suction.
\end{itemize}
\begin{itemize}[itemsep=0pt, topsep=0pt, label=\halfcircle{lightred}{lightred}]
    \item \textbf{Experimental Rigs:} Designed a Mach 3 supersonic wind tunnel to study \textit{protuberance-induced 3D shock-boundary layer interaction}. Integrated a suction chamber for passive flow control. Developed a vernier caliper-assisted precision Pitot tube for boundary layer characterization. Arranged flow visualization setup.\\
\end{itemize}
